# Ranges and Coverage Rendering Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `setRanges` (snapshot of `[start, end)` tuples, any sizes) to `d3-blockmap` with O(ranges + bins) counting, plus a `partial: "coverage"` style that blends missing to complete so range edges render crisp.

**Architecture:** A pure `ranges.ts` normalises input (clamp, sort, merge). `bitset.fill` and `bins.addRange` apply merged ranges in bulk. `state.replaceRanges` composes them. `createColorScale` gains a partial style; `pieceMap` and the React wrapper expose the new option and method.

**Tech Stack:** Same as the existing package: TypeScript strict, Vitest + happy-dom, d3-color, d3-interpolate, oxlint, oxfmt, tsup.

## Global Constraints

- `end` is exclusive in every tuple. Tuples with `end <= start` after clamping are dropped. Non-integer endpoints drop the tuple. Overlapping and touching tuples merge.
- A `Uint32Array` input is flat pairs `[s0, e0, s1, e1, ...]`; an odd trailing element is ignored.
- Counting cost for `replaceRanges` is O(ranges log ranges + bins touched); the bitset is updated with byte-level fills, never a per-block loop.
- `partial` defaults to `"gradient"` (existing behaviour). `"coverage"` interpolates `missing` to `complete`. Endpoints stay exact; the 1..254 clamp stays.
- Existing behaviour (`setBlocks`, `addBlocks`, discrete under-width mode, tooltip) is unchanged. All 54 existing tests keep passing.
- TypeScript strict. Named exports only. `import type` for type-only imports. No `any`. No magic numbers outside `src/constants.ts` (module-local derived constants tolerated). camelCase except constants. Factories not classes. No inline end-of-line comments. Block byte sizes appear nowhere.
- Run `npm run typecheck && npm run lint && npm run format && npm test` before every commit. Never run `npx biome`.

```ts
import { RANGE_PAIR } from "./constants";

export type BlockRangeTuple = readonly [start: number, end: number];
export type RangeInput = ArrayLike<BlockRangeTuple> | Uint32Array;

type Pair = [number, number];

const readPairs = (input: RangeInput): Pair[] => {
  const pairs: Pair[] = [];
  if (input instanceof Uint32Array) {
    const pairCount = Math.floor(input.length / RANGE_PAIR);
    for (let i = 0; i < pairCount; i += 1) {
      pairs.push([input[i * RANGE_PAIR], input[i * RANGE_PAIR + 1]]);
    }
    return pairs;
  }
  for (let i = 0; i < input.length; i += 1) {
    const tuple = input[i];
    pairs.push([tuple[0], tuple[1]]);
  }
  return pairs;
};

export const normalizeRanges = (input: RangeInput, totalBlocks: number): Uint32Array => {
  const clamped: Pair[] = [];
  for (const [rawStart, rawEnd] of readPairs(input)) {
    if (!Number.isInteger(rawStart) || !Number.isInteger(rawEnd)) continue;
    const start = Math.max(0, rawStart);
    const end = Math.min(totalBlocks, rawEnd);
    if (end > start) clamped.push([start, end]);
  }
  clamped.sort((a, b) => a[0] - b[0]);

  const merged: number[] = [];
  for (const [start, end] of clamped) {
    const lastIndex = merged.length - 1;
    if (merged.length > 0 && start <= merged[lastIndex]) {
      merged[lastIndex] = Math.max(merged[lastIndex], end);
    } else {
      merged.push(start, end);
    }
  }
  return Uint32Array.from(merged);
};

export const rangesLength = (flat: Uint32Array): number => {
  let total = 0;
  for (let i = 0; i < flat.length; i += RANGE_PAIR) total += flat[i + 1] - flat[i];
  return total;
};
```

```ts
import { BITS_PER_BYTE, BYTE_MASK, BYTE_SHIFT, FULL_BYTE } from "./constants";

export type Bitset = {
  has(index: number): boolean;
  set(index: number): boolean;
  fill(start: number, end: number): void;
  clear(): void;
  setCount(next: number): void;
  readonly count: number;
};

const maskRange = (from: number, to: number): number => ((1 << to) - 1) & ~((1 << from) - 1);

export const createBitset = (size: number): Bitset => {
  const bytes = new Uint8Array(Math.ceil(size / BITS_PER_BYTE));
  let count = 0;

  return {
    has: (index) => (bytes[index >>> BYTE_SHIFT] & (1 << (index & BYTE_MASK))) !== 0,
    set: (index) => {
      const byte = index >>> BYTE_SHIFT;
      const mask = 1 << (index & BYTE_MASK);
      if ((bytes[byte] & mask) !== 0) return false;
      bytes[byte] |= mask;
      count += 1;
      return true;
    },
    fill: (start, end) => {
      if (end <= start) return;
      const firstByte = start >>> BYTE_SHIFT;
      const lastByte = (end - 1) >>> BYTE_SHIFT;
      const fromBit = start & BYTE_MASK;
      const toBit = ((end - 1) & BYTE_MASK) + 1;
      if (firstByte === lastByte) {
        bytes[firstByte] |= maskRange(fromBit, toBit);
        return;
      }
      bytes[firstByte] |= maskRange(fromBit, BITS_PER_BYTE);
      bytes.fill(FULL_BYTE, firstByte + 1, lastByte);
      bytes[lastByte] |= maskRange(0, toBit);
    },
    clear: () => {
      bytes.fill(0);
      count = 0;
    },
    setCount: (next) => {
      count = next;
    },
    get count() {
      return count;
    },
  };
};
```

```ts
export type BlockRange = { start: number; end: number };

export type Bins = {
  readonly binCount: number;
  readonly counts: Uint32Array;
  binOf(index: number): number;
  rangeOf(bin: number): BlockRange;
  sizeOf(bin: number): number;
  rebuild(has: (index: number) => boolean): void;
  addRange(start: number, end: number): void;
};

export const createBins = (totalBlocks: number, binCount: number): Bins => {
  const counts = new Uint32Array(binCount);
  const binOf = (index: number): number => Math.floor((index * binCount) / totalBlocks);
  const rangeOf = (bin: number): BlockRange => ({
    start: Math.ceil((bin * totalBlocks) / binCount),
    end: Math.ceil(((bin + 1) * totalBlocks) / binCount),
  });
  const sizeOf = (bin: number): number => {
    const { start, end } = rangeOf(bin);
    return end - start;
  };

  return {
    binCount,
    counts,
    binOf,
    rangeOf,
    sizeOf,
    rebuild: (has) => {
      counts.fill(0);
      if (binCount === 0) return;
      for (let index = 0; index < totalBlocks; index += 1) {
        if (has(index)) counts[binOf(index)] += 1;
      }
    },
    addRange: (start, end) => {
      if (binCount === 0 || end <= start) return;
      const first = binOf(start);
      const last = binOf(end - 1);
      if (first === last) {
        counts[first] += end - start;
        return;
      }
      counts[first] += rangeOf(first).end - start;
      for (let bin = first + 1; bin < last; bin += 1) counts[bin] += sizeOf(bin);
      counts[last] += end - rangeOf(last).start;
    },
  };
};
```

```ts
export const createColorScale = (theme: Theme, partial: PartialStyle = "gradient"): ColorScale => {
  const table = new Uint8ClampedArray(COLOR_TABLE_SIZE * BYTES_PER_PIXEL);
  const missing = parseOr(theme.missing, DEFAULT_THEME.missing);
  const complete = parseOr(theme.complete, DEFAULT_THEME.complete);
  const partialScale =
    partial === "coverage"
      ? interpolateRgb(missing, complete)
      : interpolateRgb(
          parseOr(theme.partialLow, DEFAULT_THEME.partialLow),
          parseOr(theme.partialHigh, DEFAULT_THEME.partialHigh),
        );

  writeColor(table, 0, missing);
  writeColor(table, LAST_INDEX, complete);
  for (let i = FIRST_PARTIAL_INDEX; i <= LAST_PARTIAL_INDEX; i += 1) {
    writeColor(table, i, partialScale(i / LAST_INDEX));
  }

  return (fraction) => {
    let index = Math.round(fraction * LAST_INDEX);
    if (fraction > 0 && index < FIRST_PARTIAL_INDEX) index = FIRST_PARTIAL_INDEX;
    if (fraction < 1 && index > LAST_PARTIAL_INDEX) index = LAST_PARTIAL_INDEX;
    const offset = index * BYTES_PER_PIXEL;
    return table.subarray(offset, offset + BYTES_PER_PIXEL);
  };
};
```

```tsx
import { forwardRef, useEffect, useImperativeHandle, useRef } from "react";
import type { CSSProperties } from "react";
import type { PartialStyle } from "../colors";
import { createPieceMap } from "../pieceMap";
import type { PieceMap as PieceMapInstance, PieceMapOptions } from "../pieceMap";
import type { RangeInput } from "../ranges";

export type PieceMapProps = {
  totalBlocks: number;
  blocks?: ArrayLike<number>;
  ranges?: RangeInput;
  partial?: PartialStyle;
  className?: string;
  style?: CSSProperties;
};

export type PieceMapHandle = Pick<PieceMapInstance, "addBlocks" | "getProgress">;

const EMPTY_BLOCKS: readonly number[] = [];
const EMPTY_PROGRESS = { completed: 0, total: 0 };

export const PieceMap = forwardRef<PieceMapHandle, PieceMapProps>(
  ({ totalBlocks, blocks, ranges, partial, className, style }, ref) => {
    const containerRef = useRef<HTMLDivElement>(null);
    const instanceRef = useRef<PieceMapInstance>(null);

    useEffect(() => {
      const container = containerRef.current;
      if (!container) return undefined;
      const options: PieceMapOptions = partial ? { totalBlocks, partial } : { totalBlocks };
      const instance = createPieceMap(container, options);
      instanceRef.current = instance;
      return () => {
        instance.destroy();
        instanceRef.current = null;
      };
    }, [totalBlocks, partial]);

    useEffect(() => {
      const instance = instanceRef.current;
      if (!instance) return;
      if (ranges !== undefined) {
        instance.setRanges(ranges);
        if (blocks !== undefined) instance.addBlocks(blocks);
        return;
      }
      instance.setBlocks(blocks ?? EMPTY_BLOCKS);
    }, [totalBlocks, partial, ranges, blocks]);

    useImperativeHandle(
      ref,
      () => ({
        addBlocks: (indices) => instanceRef.current?.addBlocks(indices),
        getProgress: () =>
          instanceRef.current?.getProgress() ?? { ...EMPTY_PROGRESS, total: totalBlocks },
      }),
      [totalBlocks],
    );

    return <div ref={containerRef} className={className} style={style} />;
  },
);

PieceMap.displayName = "PieceMap";
```
